import tkinter as tk from tkinter import ttk, messagebox import json import os # اسم ملف حفظ البيانات DATA_FILE = "players_data.json" # قاعدة بيانات اللاعبين والباسوردات بتاعتهم PLAYERS_AUTH = { "مروان": "2410", "آية": "6678", "حبيبة": "6654", "ياسين": "1278" } # تحميل البيانات أو إنشاء بيانات جديدة للاعبين المحددين def load_data(): if os.path.exists(DATA_FILE): with open(DATA_FILE, "r", encoding="utf-8") as f: existing_data = json.load(f) # التأكد أن كل اللاعبين الحاليين موجودين في الملف for player in PLAYERS_AUTH: if player not in existing_data: existing_data[player] = {"tasks": [], "money": 0} return existing_data # لو الملف مش موجود، بنعمل هيكل جديد بالأسماء المطلوبة return {player: {"tasks": [], "money": 0} for player in PLAYERS_AUTH} def save_data(): with open(DATA_FILE, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=4) # تحميل البيانات data = load_data() current_user = None # --- الوظائف البرمجية --- def check_login(): """التحقق من الباسورد قبل فتح البيانات""" global current_user selected_user = user_combobox.get() entered_password = password_entry.get().strip() if not selected_user: messagebox.showwarning("تنبيه", "من فضلك اختر اسم اللاعب أولاً!") return # التأكد من صحة الباسورد if PLAYERS_AUTH.get(selected_user) == entered_password: current_user = selected_user password_entry.delete(0, tk.END) # إظهار واجهة المهام وإخفاء قفل الدخول status_label.config(text=f"أهلاً بك يا {current_user} 👋", fg="#10ac84") main_app_frame.pack(fill="both", expand=True, pady=10) update_task_list() update_money_display() else: messagebox.showerror("خطأ", "الباسورد غير صحيح! حاول مرة أخرى.") # إخفاء واجهة المهام لو الباسورد غلط main_app_frame.pack_forget() current_user = None update_money_display() def add_task(): """إضافة مهمة جديدة للمستخدم الحالي""" if not current_user: return task_text = task_entry.get().strip() if task_text: data[current_user]["tasks"].append({"name": task_text, "done": False, "paid": False}) save_data() task_entry.delete(0, tk.END) update_task_list() else: messagebox.showwarning("تنبيه", "اكتب اسم المهمة أولاً!") def toggle_task(index): """تغيير حالة المهمة وحساب الفلوس""" if current_user: task = data[current_user]["tasks"][index] task["done"] = not task["done"] # إضافة 10 جنيه لو اكتملت، وخصمها لو اتلغت if task["done"] and not task["paid"]: data[current_user]["money"] += 10 task["paid"] = True elif not task["done"] and task["paid"]: data[current_user]["money"] = max(0, data[current_user]["money"] - 10) task["paid"] = False save_data() update_task_list() update_money_display() def update_task_list(): """تحديث قائمة المهام المعروضة""" for widget in scrollable_frame.winfo_children(): widget.destroy() if not current_user: return for i, task in enumerate(data[current_user]["tasks"]): task_frame = tk.Frame(scrollable_frame, bg="#ffffff", pady=5) task_frame.pack(fill="x", expand=True, pady=2) chk_text = f"✓ {task['name']}" if task["done"] else task["name"] chk_color = "#888888" if task["done"] else "#2b2b2b" btn = tk.Button(task_frame, text=chk_text, font=("Segoe UI", 11), fg=chk_color, bg="#f0f0f0", relief="flat", anchor="w", padx=10, command=lambda idx=i: toggle_task(idx)) btn.pack(fill="x", expand=True, side="left") def update_money_display(): """تحديث رصيد الفلوس""" if current_user: total_money = data[current_user]["money"] money_label.config(text=f"إجمالي المحفظة: {total_money} جنيه") else: money_label.config(text="إجمالي المحفظة: 0 جنيه") # --- واجهة المستخدم (UI) --- root = tk.Tk() root.title("تطبيق المهام والمحفظة") root.geometry("400x650") root.configure(bg="#f5f6fa") # 1. هيدر تسجيل الدخول login_frame = tk.Frame(root, bg="#4a69bd", padx=15, pady=15) login_frame.pack(fill="x") # اختيار اللاعب user_label = tk.Label(login_frame, text="اختر اللاعب:", font=("Segoe UI", 11, "bold"), fg="#ffffff", bg="#4a69bd") user_label.pack(anchor="w") user_combobox = ttk.Combobox(login_frame, values=list(PLAYERS_AUTH.keys()), font=("Segoe UI", 11), state="readonly") user_combobox.pack(fill="x", pady=(5, 10)) # إدخال الباسورد pass_label = tk.Label(login_frame, text="أدخل الباسورد:", font=("Segoe UI", 11, "bold"), fg="#ffffff", bg="#4a69bd") pass_label.pack(anchor="w") password_entry = ttk.Entry(login_frame, font=("Segoe UI", 11), show="*") # الباسورد يظهر كـ نجوم password_entry.pack(fill="x", pady=(5, 10)) # زر الدخول login_btn = tk.Button(login_frame, text="تسجيل الدخول", font=("Segoe UI", 11, "bold"), bg="#10ac84", fg="#ffffff", relief="flat", command=check_login) login_btn.pack(fill="x", pady=5) # حالة الدخول (رسالة ترحيب أو تنبيه) status_label = tk.Label(root, text="برجاء تسجيل الدخول لعرض المهام", font=("Segoe UI", 11, "italic"), fg="#7f8c8d", bg="#f5f6fa") status_label.pack(pady=5) # --- 2. إطار التطبيق الرئيسي (مخفي حتى تسجيل الدخول الناجح) --- main_app_frame = tk.Frame(root, bg="#f5f6fa") # عرض الفلوس money_frame = tk.Frame(main_app_frame, bg="#60a3bc", padx=10, pady=10) money_frame.pack(fill="x", pady=5) money_label = tk.Label(money_frame, text="إجمالي المحفظة: 0 جنيه", font=("Segoe UI", 13, "bold"), fg="#ffffff", bg="#60a3bc") money_label.pack() # إضافة مهمة input_frame = tk.Frame(main_app_frame, bg="#f5f6fa", padx=15) input_frame.pack(fill="x", pady=5) task_entry = ttk.Entry(input_frame, font=("Segoe UI", 11)) task_entry.pack(side="left", fill="x", expand=True, padx=(0, 5)) add_btn = tk.Button(input_frame, text="إضافة مهمة", font=("Segoe UI", 10, "bold"), bg="#2980b9", fg="#ffffff", relief="flat", padx=10, command=add_task) add_btn.pack(side="right") # قائمة المهام مع السكرول list_frame = tk.Frame(main_app_frame, bg="#ffffff", bd=1, relief="groove") list_frame.pack(fill="both", expand=True, padx=15, pady=10) canvas = tk.Canvas(list_frame, bg="#ffffff", highlightthickness=0) scrollbar = ttk.Scrollbar(list_frame, orient="vertical", command=canvas.yview) scrollable_frame = tk.Frame(canvas, bg="#ffffff") scrollable_frame.bind("", lambda e: canvas.configure(scrollregion=canvas.bbox("all"))) canvas.create_window((0, 0), window=scrollable_frame, anchor="nw") canvas.configure(yscrollcommand=scrollbar.set) canvas.pack(side="left", fill="both", expand=True, padx=5, pady=5) scrollbar.pack(side="right", fill="y") # تشغيل البرنامج root.mainloop()
